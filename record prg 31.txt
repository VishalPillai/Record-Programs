FROM GARMENT
GROUP BY FCODE
B)SELECT TYPE, COUNT(GCODE)
FROM FABRIC G, GARMENT G
WHERE F.FCODE=G.FCODE
GROUP BY TYPE;
C)UPDATE GARMENT
SET PRICE = PRICE +150
WHERE DESCRIPTION = 'SLACKS';
D)SELECT G.FCODE,DESCRIPTION,PRICE,TYPE
FROM GARMENT G, FABRIC F
WHEREG.FCODE = F.FCODE;
E)SELECT MAX(FCODE),MIN(READYDATE)
FROM GARMENT
F)SELECT DISTINCT FCODE
FROM GARMENT;
G)SELECT COUNT (DISTINCT FCODE)
FROM GARMENT

FROM GARMENT
WHERE LENGTH(DESCRIPTION < 7);
